## Part two: Bison shows you how to use `Alamofire`.

```css
import UIKit

import Alamofire

class ViewController: UIViewController {

    let BaiduURL = "http://apis.haoservice.com/lifeservice/cook/query?"

override func viewDidLoad() {
    super.viewDidLoad()
    //网络请求
    self .reloadData()
}

func reloadData(){


    let parameters = [
    "menu": "土豆",
    "pn":  1,
    "rn": "10",
    "key": "2ba215a3f83b4b898d0f6fdca4e16c7c",
    ]


    Alamofire.request(.POST, BaiduURL, parameters:parameters ).responseJSON {response in

    //            debugPrint(response)
    switch response.result {
        case .Success:
            //把得到的JSON数据转为字典
            if let j = response.result.value as? NSDictionary{
                //获取字典里面的key为数组
                let Items = j.valueForKey("result")as! NSArray
            //便利数组得到每一个字典模型
            for dict in Items{

                print(dict)
            }

        }
        case .Failure(let error):

            print(error)
        }


        }

    }

}
```
